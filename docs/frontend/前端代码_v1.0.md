### 3.3 API 封装 (src/api/index.js)

```javascript
import axios from 'axios'
import { ElMessage } from 'element-plus'

const api = axios.create({
  baseURL: '/api',
  timeout: 30000,
})

// 请求拦截器
api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token')
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 响应拦截器
api.interceptors.response.use(
  (response) => {
    if (response.data.code === 200) {
      return response.data.data
    } else {
      ElMessage.error(response.data.message || '请求失败')
      return Promise.reject(new Error(response.data.message))
    }
  },
  (error) => {
    ElMessage.error(error.message || '网络错误')
    return Promise.reject(error)
  }
)

export default api
```

### 3.4 访谈 API (src/api/interviews.js)

```javascript
import api from './index'

export const interviewsApi = {
  // 获取访谈列表
  getList(params) {
    return api.get('/interviews', { params })
  },

  // 获取访谈详情
  getById(id) {
    return api.get(`/interviews/${id}`)
  },

  // 创建访谈
  create(data) {
    return api.post('/interviews', data)
  },

  // 更新访谈
  update(id, data) {
    return api.put(`/interviews/${id}`, data)
  },

  // 删除访谈
  delete(id) {
    return api.delete(`/interviews/${id}`)
  },

  // 开始访谈
  start(id) {
    return api.post(`/interviews/${id}/start`)
  },

  // 完成访谈
  complete(id) {
    return api.post(`/interviews/${id}/complete`)
  },

  // 采纳追问建议
  adoptSuggestion(id, suggestionId) {
    return api.post(`/interviews/${id}/suggestions/${suggestionId}/adopt`)
  },

  // 跳过追问建议
  skipSuggestion(id, suggestionId) {
    return api.post(`/interviews/${id}/suggestions/${suggestionId}/skip`)
  },

  // 刷新追问建议
  refreshSuggestions(id) {
    return api.post(`/interviews/${id}/suggestions/refresh`)
  },
}
```

## 4. 状态管理 (Pinia)

### 4.1 访谈状态 (src/stores/interview.js)

```javascript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { interviewsApi } from '@/api/interviews'

export const useInterviewStore = defineStore('interview', () => {
  // 状态
  const currentInterview = ref(null)
  const transcripts = ref([])
  const summaries = ref([])
  const suggestions = ref([])
  const gaps = ref([])
  const isRecording = ref(false)
  const sseConnection = ref(null)

  // 计算属性
  const activeSuggestion = computed(() => {
    return suggestions.value.find(s => s.status === 1) || null
  })

  // 方法
  async function fetchInterview(id) {
    currentInterview.value = await interviewsApi.getById(id)
    return currentInterview.value
  }

  async function createInterview(data) {
    return await interviewsApi.create(data)
  }

  async function startInterview(id) {
    const result = await interviewsApi.start(id)
    connectSSE(id, result.sseUrl)
    return result
  }

  function connectSSE(interviewId, url) {
    const eventSource = new EventSource(url)

    eventSource.addEventListener('transcript', (e) => {
      const data = JSON.parse(e.data)
      transcripts.value.push({
        id: Date.now(),
        ...data,
      })
    })

    eventSource.addEventListener('summary', (e) => {
      const data = JSON.parse(e.data)
      summaries.value.push(data)
    })

    eventSource.addEventListener('suggestion', (e) => {
      const data = JSON.parse(e.data)
      suggestions.value.push({
        ...data,
        status: 1, // 待处理
      })
    })

    eventSource.addEventListener('gap', (e) => {
      const data = JSON.parse(e.data)
      gaps.value.push(data)
    })

    eventSource.onerror = () => {
      console.error('SSE connection error')
      eventSource.close()
    }

    sseConnection.value = eventSource
  }

  function closeSSE() {
    if (sseConnection.value) {
      sseConnection.value.close()
      sseConnection.value = null
    }
  }

  async function adoptSuggestion(suggestionId) {
    await interviewsApi.adoptSuggestion(currentInterview.value.id, suggestionId)
    const idx = suggestions.value.findIndex(s => s.id === suggestionId)
    if (idx !== -1) {
      suggestions.value[idx].status = 2
    }
  }

  async function skipSuggestion(suggestionId) {
    await interviewsApi.skipSuggestion(currentInterview.value.id, suggestionId)
    const idx = suggestions.value.findIndex(s => s.id === suggestionId)
    if (idx !== -1) {
      suggestions.value[idx].status = 3
    }
  }

  function reset() {
    currentInterview.value = null
    transcripts.value = []
    summaries.value = []
    suggestions.value = []
    gaps.value = []
    isRecording.value = false
    closeSSE()
  }

  return {
    // 状态
    currentInterview,
    transcripts,
    summaries,
    suggestions,
    gaps,
    isRecording,
    // 计算属性
    activeSuggestion,
    // 方法
    fetchInterview,
    createInterview,
    startInterview,
    closeSSE,
    adoptSuggestion,
    skipSuggestion,
    reset,
  }
})
```

### 6.1 SSE 工具 (src/utils/sse.js)

```javascript
export function createSSE(url, handlers) {
  const eventSource = new EventSource(url)

  if (handlers.transcript) {
    eventSource.addEventListener('transcript', (e) => {
      handlers.transcript(JSON.parse(e.data))
    })
  }

  if (handlers.summary) {
    eventSource.addEventListener('summary', (e) => {
      handlers.summary(JSON.parse(e.data))
    })
  }

  if (handlers.suggestion) {
    eventSource.addEventListener('suggestion', (e) => {
      handlers.suggestion(JSON.parse(e.data))
    })
  }

  eventSource.onerror = (e) => {
    console.error('SSE error:', e)
    eventSource.close()
    handlers.error?.(e)
  }

  return {
    close: () => eventSource.close(),
  }
}
```

### 6.2 音频录制工具 (src/utils/audio.js)

```javascript
export class AudioRecorder {
  constructor() {
    this.mediaRecorder = null
    this.chunks = []
    this.isRecording = false
  }

  async start() {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
    this.mediaRecorder = new MediaRecorder(stream)
    this.chunks = []

    this.mediaRecorder.ondataavailable = (e) => {
      if (e.data.size > 0) {
        this.chunks.push(e.data)
      }
    }

    this.mediaRecorder.start()
    this.isRecording = true
  }

  stop() {
    return new Promise((resolve) => {
      this.mediaRecorder.onstop = () => {
        const blob = new Blob(this.chunks, { type: 'audio/webm' })
        resolve(blob)
      }
      this.mediaRecorder.stop()
      this.isRecording = false
    })
  }

  pause() {
    this.mediaRecorder.pause()
  }

  resume() {
    this.mediaRecorder.resume()
  }
}
```
