## 4. 中间件

### 4.1 认证中间件 (src/middleware/auth.js)

```javascript
const jwt = require('jsonwebtoken')
const config = require('../config')
const { PrismaClient } = require('@prisma/client')

const prisma = new PrismaClient()

const auth = async (req, res, next) => {
  try {
    const authHeader = req.headers.authorization
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return res.status(401).json({ code: 401, message: '未提供认证令牌' })
    }

    const token = authHeader.split(' ')[1]
    const decoded = jwt.verify(token, config.jwt.secret)

    const user = await prisma.user.findUnique({
      where: { userId: decoded.userId },
    })

    if (!user) {
      return res.status(401).json({ code: 401, message: '用户不存在' })
    }

    req.user = user
    next()
  } catch (error) {
    if (error.name === 'JsonWebTokenError') {
      return res.status(401).json({ code: 401, message: '认证令牌无效' })
    }
    next(error)
  }
}

module.exports = auth
```

### 4.2 错误处理中间件 (src/middleware/error.js)

```javascript
const errorMiddleware = (err, req, res, next) => {
  console.error('Error:', err)

  const status = err.status || 500
  const message = err.message || '内部服务器错误'

  res.status(status).json({
    code: status,
    message,
    ...(process.env.NODE_ENV === 'development' && { stack: err.stack }),
  })
}

module.exports = errorMiddleware
```

### 4.3 文件上传中间件 (src/middleware/upload.js)

```javascript
const multer = require('multer')
const path = require('path')

const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    cb(null, 'uploads/')
  },
  filename: (req, file, cb) => {
    const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9)
    cb(null, uniqueSuffix + path.extname(file.originalname))
  },
})

const upload = multer({
  storage,
  limits: {
    fileSize: 100 * 1024 * 1024, // 100MB
  },
  fileFilter: (req, file, cb) => {
    const allowedTypes = /audio|video|application/
    const extname = allowedTypes.test(path.extname(file.originalname).toLowerCase())
    const mimetype = allowedTypes.test(file.mimetype)

    if (extname || mimetype) {
      cb(null, true)
    } else {
      cb(new Error('不支持的文件类型'))
    }
  },
})

module.exports = upload
```

## 6. 控制器层

### 6.1 认证控制器 (src/controllers/auth.controller.js)

```javascript
const jwt = require('jsonwebtoken')
const { PrismaClient } = require('@prisma/client')
const config = require('../config')

const prisma = new PrismaClient()

const login = async (req, res, next) => {
  try {
    const { userId, name } = req.body

    if (!userId || !name) {
      return res.status(400).json({ code: 400, message: '工号和姓名为必填项' })
    }

    // 查找或创建用户
    let user = await prisma.user.findUnique({ where: { userId } })

    if (!user) {
      user = await prisma.user.create({
        data: { userId, name },
      })
    }

    // 生成 JWT
    const token = jwt.sign(
      { userId: user.userId, id: user.id },
      config.jwt.secret,
      { expiresIn: config.jwt.expiresIn }
    )

    res.json({
      code: 200,
      data: {
        token,
        user: {
          id: user.id,
          userId: user.userId,
          name: user.name,
          email: user.email,
        },
      },
    })
  } catch (error) {
    next(error)
  }
}

const logout = async (req, res, next) => {
  try {
    // MVP 阶段简单返回成功
    res.json({ code: 200, message: '登出成功' })
  } catch (error) {
    next(error)
  }
}

const getCurrentUser = async (req, res, next) => {
  try {
    res.json({
      code: 200,
      data: req.user,
    })
  } catch (error) {
    next(error)
  }
}

module.exports = {
  login,
  logout,
  getCurrentUser,
}
```

### 6.2 访谈控制器 (src/controllers/interview.controller.js)

```javascript
const { PrismaClient } = require('@prisma/client')
const interviewService = require('../services/interview.service')
const { getSSemanager } = require('../sse/sseManager')

const prisma = new PrismaClient()

const getList = async (req, res, next) => {
  try {
    const { projectId, keyword, status, page = 1, pageSize = 20 } = req.query

    const result = await interviewService.getList({
      projectId: projectId ? BigInt(projectId) : null,
      keyword,
      status: status ? parseInt(status) : null,
      page: parseInt(page),
      pageSize: parseInt(pageSize),
      creatorId: req.user.id,
    })

    res.json({ code: 200, data: result })
  } catch (error) {
    next(error)
  }
}

const getById = async (req, res, next) => {
  try {
    const interview = await interviewService.getById(req.params.id, req.user.id)
    res.json({ code: 200, data: interview })
  } catch (error) {
    next(error)
  }
}

const create = async (req, res, next) => {
  try {
    const { title, researchGoal, projectId, participants, audioSource, attachments } = req.body

    const interview = await interviewService.create({
      title,
      researchGoal,
      projectId: BigInt(projectId),
      creatorId: req.user.id,
      participants,
      audioSource,
      attachments,
    })

    res.json({ code: 200, data: interview })
  } catch (error) {
    next(error)
  }
}

const update = async (req, res, next) => {
  try {
    const interview = await interviewService.update(req.params.id, req.body, req.user.id)
    res.json({ code: 200, data: interview })
  } catch (error) {
    next(error)
  }
}

const del = async (req, res, next) => {
  try {
    await interviewService.del(req.params.id, req.user.id)
    res.json({ code: 200, message: '删除成功' })
  } catch (error) {
    next(error)
  }
}

const start = async (req, res, next) => {
  try {
    const sessionId = await interviewService.start(req.params.id, req.user.id)
    res.json({
      code: 200,
      data: {
        sessionId,
        sseUrl: `/api/interviews/${req.params.id}/stream`,
      },
    })
  } catch (error) {
    next(error)
  }
}

const complete = async (req, res, next) => {
  try {
    const summary = await interviewService.complete(req.params.id, req.user.id)
    res.json({ code: 200, data: { summary } })
  } catch (error) {
    next(error)
  }
}

const stream = async (req, res, next) => {
  try {
    const sseManager = getSSemanager()
    sseManager.addClient(res, req.params.id)

    req.on('close', () => {
      sseManager.removeClient(res, req.params.id)
    })
  } catch (error) {
    next(error)
  }
}

const refreshSuggestions = async (req, res, next) => {
  try {
    await interviewService.refreshSuggestions(req.params.id, req.user.id)
    res.json({ code: 200, message: '追问建议已刷新' })
  } catch (error) {
    next(error)
```

### 8.1 SSE Manager (src/sse/sseManager.js)

```javascript
const clients = new Map()

const initSSE = (app) => {
  app.use((req, res, next) => {
    res.sse = {
      send: (event, data) => {
        res.write(`event: ${event}\n`)
        res.write(`data: ${JSON.stringify(data)}\n\n`)
      },
    }
    next()
  })

  console.log('SSE 服务已初始化')
}

const	addClient = (res, interviewId) => {
  // 设置 SSE 响应头
  res.setHeader('Content-Type', 'text/event-stream')
  res.setHeader('Cache-Control', 'no-cache')
  res.setHeader('Connection', 'keep-alive')
  res.setHeader('X-Accel-Buffering', 'no')

  const clientId = `${interviewId}_${Date.now()}`
  clients.set(clientId, { res, interviewId })

  // 发送初始连接消息
  res.write(`event: connected\ndata: {"clientId": "${clientId}"}\n\n`)

  console.log(`SSE 客户端已连接：${clientId}, interviewId: ${interviewId}`)
}

const removeClient = (res, interviewId) => {
  for (const [clientId, client] of clients.entries()) {
    if (client.res === res) {
      clients.delete(clientId)
      console.log(`SSE 客户端已断开：${clientId}`)
      break
    }
  }
}

const send = (interviewId, event, data) => {
  for (const [clientId, client] of clients.entries()) {
    if (client.interviewId === interviewId) {
      try {
        client.res.write(`event: ${event}\n`)
        client.res.write(`data: ${JSON.stringify(data)}\n\n`)
      } catch (error) {
        console.error(`SSE 发送失败：${clientId}`, error)
        clients.delete(clientId)
      }
    }
  }
}

const getSSemanager = () => {
  return {
    addClient,
    removeClient,
    send,
  }
}

module.exports = {
  initSSE,
  getSSemanager,
}
```
